retry.ts
export async function retry(fn: () => Promise, tries = 3) { for (let i = 0; ; i++) { try { return await fn(); } catch (err) { if (i >= tries - 1) throw err; await sleep(2 ** i * 250); // back off: 250ms, 500ms, 1s } } }
  • Retries only on failure
  • Waits longer each time
  • Gives up after N tries
